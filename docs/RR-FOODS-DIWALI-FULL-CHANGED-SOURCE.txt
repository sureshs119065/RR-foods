        parsed = urlparse(item.youtube_url)

@bp.get("/bundles")
def bundles():
    now = datetime.now(timezone.utc)
    items = db.session.scalars(select(Bundle).where(Bundle.is_active.is_(True),
        (Bundle.starts_at.is_(None)) | (Bundle.starts_at <= now),
        (Bundle.ends_at.is_(None)) | (Bundle.ends_at > now)).order_by(Bundle.created_at.desc())).all()
    products = {p.id: p for p in db.session.scalars(select(Product).where(Product.is_deleted.is_(False))).all()}
    originals = {}
    bundle_prices = {}
    for bundle in items:
        originals[bundle.id] = sum((Decimal(str(products[int(line.get("product_id", 0))].final_price)) * int(line.get("quantity", 1))
            for line in (bundle.product_quantities or []) if int(line.get("product_id", 0)) in products), Decimal("0"))
        # Normalize database numeric values before template comparisons/subtraction.
        bundle_prices[bundle.id] = Decimal(str(bundle.bundle_price)).quantize(Decimal("0.01"))
    return render_template("shop/bundles.html", items=items, products=products, originals=originals, bundle_prices=bundle_prices)

@bp.post("/bundles/<int:bundle_id>/add")
def add_bundle(bundle_id):
    if not g.get("user") or g.user.role != "user":
        return redirect(url_for("auth.login", next=url_for("shop.bundles")))
    now = datetime.now(timezone.utc)
    bundle = db.session.get(Bundle, bundle_id)
    if not bundle or not bundle.is_active or (bundle.starts_at and bundle.starts_at > now) or (bundle.ends_at and bundle.ends_at <= now):
        abort(404)
    lines = []
    for entry in bundle.product_quantities or []:
        product = db.session.get(Product, int(entry.get("product_id", 0)))
        qty = int(entry.get("quantity", 1))
        if not product or not product.is_active or product.is_deleted or active_variants(product) or qty < 1 or product.stock_qty < qty:
            flash("A bundle product is unavailable or has insufficient stock.", "error")
            return redirect(url_for("shop.bundles"))
        lines.append((product, qty))
    regular_total = sum((Decimal(product.final_price) * qty for product, qty in lines), Decimal("0"))
    bundle_total = Decimal(str(bundle.bundle_price)).quantize(Decimal("0.01"))
    if not lines or regular_total <= 0 or bundle_total <= 0 or bundle_total > regular_total:
        flash("This bundle price is not configured correctly. Please contact the shop.", "error")
        return redirect(url_for("shop.bundles"))
    allocated = Decimal("0")
    for index, (product, qty) in enumerate(lines):
        line_total = (bundle_total - allocated) if index == len(lines)-1 else (bundle_total * (Decimal(product.final_price) * qty / regular_total)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        allocated += line_total
        unit_price = (line_total / qty).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
        item = db.session.scalar(select(CartItem).where(CartItem.user_id == g.user.id, CartItem.product_id == product.id,
            CartItem.variant_id.is_(None), CartItem.bundle_id == bundle.id, CartItem.saved_for_later.is_(False)))
        if item:
            if item.quantity + qty > product.stock_qty:
                db.session.rollback()
                flash(f"Not enough stock to add another {bundle.name}.", "error")
                return redirect(url_for("shop.bundles"))
            item.quantity += qty
            item.line_total_override = (item.unit_price_override or unit_price) * item.quantity
        else:
            db.session.add(CartItem(user_id=g.user.id, product_id=product.id, variant_id=None, bundle_id=bundle.id,
                unit_price_override=unit_price, line_total_override=line_total, quantity=qty))
    db.session.commit()
    flash(f"Added {bundle.name} to your cart at its bundle price.", "success")
    return redirect(url_for("shop.cart"))

@bp.get("/banner-colors.css")
def banner_colors():
    """Per-banner colours as real CSS rules. The site's CSP forbids style="" attributes, and this keeps the first paint correct without JavaScript."""
    rules = []
    for b in header_banners.cached_payload()["banners"]:
        bg = b["bg_color"] if b.get("bg_color") and HEX.match(b["bg_color"]) else "#1b5e3b"
        fg = b["text_color"] if b.get("text_color") and HEX.match(b["text_color"]) else "#ffffff"
        rules.append(f".hb-{int(b['id'])}{{--hb-bg:{bg};--hb-fg:{fg}}}")
    resp = make_response("\n".join(rules) or "/* no banners */")
    resp.mimetype = "text/css"
    resp.headers["Cache-Control"] = "public, max-age=60"
    return resp


def _render_listing(*, slug=None, bestseller_only=False):
    f = Filters.from_args(request.args)
    if slug:
        f.category = slug
    page, cat, brands = storefront.listing(f, bestseller_only=bestseller_only)
    if f.category and cat is None:
        abort(404)

    # Keep all active filters in links and pagination; never carry a page number
    # into a newly submitted filter form.
    query_args = {k: v for k, v in request.args.lists()
                  if k not in ("page", "bestsellers", "category") and any(v)}
    if cat:
        if slug:
            query_args["slug"] = cat.slug
        else:
            query_args["category"] = cat.slug

    if bestseller_only:
        title = "Our Bestsellers"
        crumbs = [("Home", "/"), ("Bestsellers", None)]
        endpoint = "shop.bestsellers"
        canonical = url_for("shop.bestsellers", _external=True)
    elif cat:
        title = cat.name
        crumbs = storefront.breadcrumb_for(cat)
        endpoint = "shop.category_listing"
        canonical = url_for("shop.category_listing", slug=cat.slug, _external=True)
    elif f.q:
        title = f"Search results for “{f.q}”"
        crumbs = [("Home", "/"), ("Search results", None)]
        endpoint = "shop.listing"
        canonical = url_for("shop.listing", _external=True)
    else:
        title = "Shop All Products"
        crumbs = [("Home", "/"), ("Shop All", None)]
        endpoint = "shop.listing"
        canonical = url_for("shop.listing", _external=True)

    clear_args = {"q": f.q} if f.q else {}
    clean = not (f.q or f.active_count() or request.args.get("sort"))
    subcats = (db.session.scalars(
        select(Category).where(Category.parent_id == cat.id, Category.is_deleted.is_(False),
                               Category.is_active.is_(True)).order_by(Category.sort_order, Category.name)
    ).all() if cat else [])
    return render_template(
        "shop/listing.html", f=f, page=page, cat=cat, brands=brands,
        sorts=storefront.SORTS, crumbs=crumbs, title=title, listing_endpoint=endpoint,
        listing_args=query_args, canonical=canonical, clear_args=clear_args,
        listing_url=(url_for(endpoint, slug=cat.slug) if cat and endpoint == "shop.category_listing"
                     else url_for(endpoint)),
        clear_url=(url_for(endpoint, slug=cat.slug, **clear_args) if cat and endpoint == "shop.category_listing"
                   else url_for(endpoint, category=cat.slug, **clear_args) if cat
                   else url_for(endpoint, **clear_args)),
        index_ok=clean and f.page == 1, subcats=subcats,
        bestseller_only=bestseller_only,
    )

@bp.get("/product/<slug>")
def product(slug):
    p = storefront.get_product(slug)
    if p is None:
        abort(404)
    storefront.remember_view(p.id)
    now = utcnow()
    offers = [o for o in p.offers if o.is_active and (not o.starts_at or o.starts_at <= now) and (not o.ends_at or o.ends_at > now)]
    coupons = db.session.scalars(select(Coupon).where(Coupon.is_active.is_(True), (Coupon.expires_at.is_(None)) | (Coupon.expires_at > now)).order_by(Coupon.id).limit(3)).all()
    reviews = db.session.scalars(select(Review).where(Review.product_id == p.id, Review.status == "approved", Review.is_deleted.is_(False))
                                 .options(joinedload(Review.user)).order_by(Review.helpful_count.desc(), Review.created_at.desc()).limit(10)).all()
    questions = db.session.scalars(select(ProductQuestion).where(ProductQuestion.product_id == p.id, ProductQuestion.is_deleted.is_(False), ProductQuestion.answer.is_not(None))
                                   .order_by(ProductQuestion.created_at.desc()).limit(10)).all()
    shopper = g.get("user") is not None and g.user.role == "user"
    can_review = bool(shopper and review_svc.purchase_for(g.user.id, p.id))
    my_review = db.session.scalar(select(Review).where(Review.user_id == g.user.id, Review.product_id == p.id, Review.is_deleted.is_(False))) if shopper else None
    voted = set(db.session.scalars(select(ReviewVote.review_id).where(ReviewVote.user_id == g.user.id, ReviewVote.review_id.in_([r.id for r in reviews] or [0])))) if shopper else set()
    cat = db.session.get(Category, p.category_id) if p.category_id else None
    crumbs = storefront.breadcrumb_for(cat)
    if cat:
        crumbs[-1] = (cat.name, f"/shop?category={cat.slug}")
    crumbs.append((p.name, None))
    variants = active_variants(p)
    together = storefront.bought_together(p)
    from ..models import Bundle
    bundles = db.session.scalars(select(Bundle).where(Bundle.is_active.is_(True))).all()
    product_bundles = [b for b in bundles if any(int(x.get("product_id", 0)) == p.id for x in (b.product_quantities or []))]
    return render_template("shop/product.html", p=p, variants=variants, stock=available_qty(p), offers=offers, coupons=coupons, reviews=reviews,
                           questions=questions, crumbs=crumbs, together=together, together_ids=[t.id for t in together if not active_variants(t)],
                           similar=storefront.you_may_like(p), shopper=shopper, can_review=can_review, my_review=my_review, voted=voted, recent=storefront.recently_viewed(exclude=p.id), now=now, product_bundles=product_bundles)

@bp.get("/api/pincode/<pin>")
@limiter.limit("60 per minute")
def api_pincode(pin):
    if not re.fullmatch(r"\d{6}", pin):
        return jsonify(ok=False, message="Enter a 6 digit pincode."), 422
    zone = db.session.get(DeliveryZone, pin)
    if zone is None or not zone.is_serviceable:
        return jsonify(ok=True, deliverable=False, message=f"Sorry, we do not deliver to {pin} yet.")
    days = zone.eta_days or 3
    fee = float(zone.delivery_fee or 0)
    return jsonify(ok=True, deliverable=True, eta_days=days, fee=fee,
                   message=f"Delivery to {pin} in about {days} day{'s' if days != 1 else ''}" + (" (free delivery)." if fee == 0 else f" (delivery ₹{fee:,.0f})."))

def _products():
    rows = db.session.scalars(select(Product).where(Product.is_deleted.is_(False), Product.is_active.is_(True), Product.stock_qty > 0).order_by(Product.name)).all()

def _save(combo=None):
    errors = {}
    products = _products()
    if request.method == "POST":
        name = clean(request.form.get("name"), 180)
        description = clean(request.form.get("description"), 1000) or None
        image = clean(request.form.get("image"), 500) or None
        festival = clean(request.form.get("festival_tag"), 40).lower() or "diwali"
        price = to_decimal(request.form.get("price"), "price", errors, min_=Decimal("0.01"), max_=Decimal("1000000"))
        original = to_decimal(request.form.get("original_price"), "original_price", errors, min_=Decimal("0.01"), max_=Decimal("1000000"))
        ids = []
        for raw in request.form.getlist("product_ids")[:12]:
            if raw.isdigit() and int(raw) not in ids:
                ids.append(int(raw))
        selected = [p for p in products if p.id in ids]
        if len(name) < 3: errors["name"] = "Enter a combo name."
        if not selected: errors["items"] = "Choose at least one active product."
        if price and original and price >= original: errors["price"] = "Combo price must be lower than original price."
        end_date = clean(request.form.get("diwali_end_at"), 30)
        if end_date:
            from datetime import datetime
            try: datetime.fromisoformat(end_date)
            except ValueError: errors["diwali_end_at"] = "Enter a valid date and time."
        if not errors:
            is_new = combo is None
            if is_new:
                slug = unique_slug(Combo, name)
                bundle = Bundle(name=name, slug=unique_slug(Bundle, f"{slug}-cart"), description=description,
                                product_quantities=[{"product_id": p.id, "quantity": 1} for p in selected],
                                bundle_price=float(price), is_active=checked("is_active"))
                db.session.add(bundle)
                db.session.flush()
                combo = Combo(name=name, slug=slug, bundle=bundle)
                db.session.add(combo)
            else:
                bundle = combo.bundle
                bundle.name, bundle.description = name, description
                bundle.product_quantities = [{"product_id": p.id, "quantity": 1} for p in selected]
                bundle.bundle_price = float(price)
                bundle.is_active = checked("is_active")
            combo.name, combo.description, combo.items = name, description, [{"product_id": p.id, "quantity": 1} for p in selected]
            combo.price, combo.original_price = price, original
            combo.image, combo.festival_tag, combo.is_active = image, festival, checked("is_active")
            if end_date:
                setting = db.session.get(SiteSetting, "diwali_end_at") or SiteSetting(key="diwali_end_at", value=end_date)
                setting.value = end_date
                db.session.add(setting)
            elif is_new and db.session.get(SiteSetting, "diwali_end_at") is None:
                db.session.add(SiteSetting(key="diwali_end_at", value="2026-11-12T23:59"))
            db.session.flush()
            log("combo.create" if is_new else "combo.update", "combo", combo.id,
                after={"name": name, "price": str(price), "is_active": combo.is_active})
            db.session.commit()
            flash(f"{name} saved.", "success")
            return redirect(url_for("admin.combos"))
        f = {"name": name, "description": description or "", "image": image or "", "festival_tag": festival,
             "price": request.form.get("price", ""), "original_price": request.form.get("original_price", ""),
             "is_active": checked("is_active"), "selected_ids": ids, "diwali_end_at": end_date}
        return render_template("admin/combo_form.html", combo=combo, f=f, products=products, errors=errors), 422

    setting = db.session.get(SiteSetting, "diwali_end_at")
    if combo:
        selected_ids = [int(x.get("product_id")) for x in (combo.items or []) if str(x.get("product_id", "")).isdigit()]
        f = {"name": combo.name, "description": combo.description or "", "image": combo.image or "",
             "festival_tag": combo.festival_tag, "price": str(combo.price), "original_price": str(combo.original_price),
             "is_active": combo.is_active, "selected_ids": selected_ids,
             "diwali_end_at": setting.value if setting and isinstance(setting.value, str) else ""}
    else:
        f = {"name": "", "description": "", "image": "", "festival_tag": "diwali", "price": "",
             "original_price": "", "is_active": True, "selected_ids": [],
             "diwali_end_at": setting.value if setting and isinstance(setting.value, str) else "2026-11-12T23:59"}
    return render_template("admin/combo_form.html", combo=combo, f=f, products=products, errors=errors)

def _register_diwali_seed(app):
    @app.cli.command("seed-diwali-combos")
    def seed_diwali_combos():
        """Create three sample Diwali combo records using existing active catalogue products."""
        from decimal import Decimal
        from sqlalchemy import select
        from .models import Bundle, Combo, Product, SiteSetting
        from .seed import slugify
        from .services.storefront import active_variants
        rows = db.session.scalars(select(Product).where(Product.is_active.is_(True), Product.is_deleted.is_(False), Product.final_price > 0, Product.stock_qty > 0).order_by(Product.name)).all()
        products = [p for p in rows if not active_variants(p)]
        if len(products) < 2:
            raise click.ClickException("Import and activate at least two products before seeding Diwali combos.")
        if db.session.scalar(select(Combo.id).limit(1)):
            click.echo("Diwali combos already exist; no duplicate samples were created.")
            return
        plans = [
            ("Biotin Laddu Festive Box", "A festive RR FOODS laddu selection.", ("laddu",)),
            ("Healthy Snack Combo", "A shareable mix of millet and wholesome pantry favourites.", ("muesli", "moringa", "soup")),
            ("Traditional Pantry Gift Hamper", "Traditional RR FOODS pantry favourites, curated for gifting.", ("idli podi", "pickle", "vathal")),
        ]
        used_ids = set()
        for name, description, keywords in plans:
            chosen = []
            for keyword in keywords:
                match = next((p for p in products if keyword in p.name.lower() and p.id not in used_ids), None)
                if match and len(chosen) < 3:
                    chosen.append(match); used_ids.add(match.id)
            if not chosen:
                chosen = [p for p in products if p.id not in used_ids][:2]
                used_ids.update(p.id for p in chosen)
            if not chosen:
                chosen = products[:2]
            original = sum((Decimal(str(p.final_price)) for p in chosen), Decimal("0"))
            if original <= 0: original = Decimal("200.00")
            price = (original * Decimal("0.85")).quantize(Decimal("0.01"))
            slug = slugify(name)
            from .admin.helpers import unique_slug
            bundle = Bundle(name=name, slug=unique_slug(Bundle, f"diwali-{slug}-cart"), description=description,
                product_quantities=[{"product_id": p.id, "quantity": 1} for p in chosen], bundle_price=float(price), is_active=True)
            db.session.add(bundle); db.session.flush()
            db.session.add(Combo(name=name, slug=f"diwali-{slug}", description=description,
                items=[{"product_id": p.id, "quantity": 1} for p in chosen], price=price,
                original_price=original, festival_tag="diwali", is_active=True, bundle=bundle))
        setting = db.session.get(SiteSetting, "diwali_end_at")
        if not setting: db.session.add(SiteSetting(key="diwali_end_at", value="2026-11-12T23:59"))
        db.session.commit()
        click.echo("Created 3 Diwali combos using existing active catalogue products.")

# RR FOODS Quality Journey and Diwali Combos

## About page
`/about` now receives an editable `quality_stages` list from `app/shop/content.py`. The About template renders the six steps and inline SVG icons. `rr-foods-journey.css` provides the responsive vertical/mobile and horizontal/desktop timeline. `quality-journey.js` reveals cards with IntersectionObserver, counts the small verified numeric metrics, draws scroll progress and moves the marker. It honours `prefers-reduced-motion` and leaves content visible without JavaScript.

The only numeric claims displayed are `0 preservatives added` (the supplied brand requirement) and `6 journey stages`. A previously unsupported `50+ quality checks` claim is deliberately not shown.

## Diwali combos
A new `Combo` table stores name, description, item/product list, discounted price, original price, image URL, active state, festival tag and its linked existing `Bundle`. The linked Bundle is used by the existing bundle/cart implementation, so cart pricing is allocated across the product lines at the combo price rather than merely adding full-price products.

Admin routes are under `/admin/combos`: list, create, edit and delete. The form controls product selection, active state, image, prices, festival tag and the site-wide Diwali countdown end time stored in `site_settings` (key: `diwali_end_at`). Only active, in-stock products without active variants are selectable, matching the existing bundle-cart constraints.

Run `flask seed-diwali-combos` after products are imported, active and stocked. It creates three sample combos from actual catalogue products and does not overwrite existing combos.

## Fireworks
`diwali-fireworks.js` uses a bounded canvas particle pool, gravity, fading trails and rocket bursts. Automatic launches run about every 1.2 seconds only while the section is visible and the tab is active. The section wrapper captures pointer clicks; canvas has `pointer-events:none`, leaving cards and forms usable. The Celebrate button fires a multi-burst show. There is no audio by default. Reduced-motion users get a static celebratory state without automatic animation.

## Setup
1. Back up the database and `.env`.
2. Install project requirements in the project's virtual environment.
3. Run `flask db upgrade` to create the `combos` table.
4. Ensure catalogue products are active and have stock; then run `flask seed-diwali-combos` for sample records.
5. Open `/admin/combos` to edit sample contents, prices, active state and countdown date.
6. Verify `/about` and `/offers` on desktop and mobile.
